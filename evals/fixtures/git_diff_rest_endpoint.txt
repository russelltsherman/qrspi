+const { UserPreferences, DEFAULT_PREFERENCES } = require('../models/user_preferences')
+const { canAccess } = require('../auth/access')
+const BaseController = require('./base')
+
+// DASH-417: read-only notification + display preferences endpoint.
+async function getPreferences(req, res) {
+  const targetId = req.params.id
+  if (!canAccess(req.auth, targetId)) {
+    return BaseController.fail(res, 403, 'forbidden', 'Cannot read another user\'s preferences')
+  }
+  const prefs = (await UserPreferences.forUser(targetId)) ?? DEFAULT_PREFERENCES
+  return res.json({
+    notifications: prefs.notifications,
+    display: prefs.display,
+  })
+}
+
+module.exports = { getPreferences }

+const request = require('supertest')
+const app = require('../../src/app')
+const { seedUser, authToken } = require('../helpers/auth')
+
+describe('GET /api/users/:id/preferences', () => {
+  it('returns notifications and display for the authenticated self', async () => {
+    const user = await seedUser({ notifications: { email: true }, display: { theme: 'dark' } })
+    const res = await request(app)
+      .get(`/api/users/${user.id}/preferences`)
+      .set('Authorization', authToken(user))
+    expect(res.status).toBe(200)
+    expect(res.body.notifications).toEqual({ email: true })
+    expect(res.body.display).toEqual({ theme: 'dark' })
+  })
+
+  it('falls back to DEFAULT_PREFERENCES when no row exists', async () => {
+    const user = await seedUser({ withPreferences: false })
+    const res = await request(app)
+      .get(`/api/users/${user.id}/preferences`)
+      .set('Authorization', authToken(user))
+    expect(res.status).toBe(200)
+    expect(res.body.notifications).toBeDefined()
+    expect(res.body.display).toBeDefined()
+  })
+
+  it('returns 401 without a token', async () => {
+    const user = await seedUser()
+    const res = await request(app).get(`/api/users/${user.id}/preferences`)
+    expect(res.status).toBe(401)
+  })
+
+  it('returns 403 for a non-admin reading another user', async () => {
+    const owner = await seedUser()
+    const other = await seedUser()
+    const res = await request(app)
+      .get(`/api/users/${owner.id}/preferences`)
+      .set('Authorization', authToken(other))
+    expect(res.status).toBe(403)
+  })
+
+  it('returns 404 for an unknown user id', async () => {
+    const admin = await seedUser({ role: 'admin' })
+    const res = await request(app)
+      .get('/api/users/does-not-exist/preferences')
+      .set('Authorization', authToken(admin))
+    expect(res.status).toBe(404)
+  })
+})

+  process.exit(passed ? 0 : 1)
+}
